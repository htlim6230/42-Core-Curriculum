#include <stdio.h>
#include <stdio.h>

char	*ft_strnstr(const char *haystack, const char *needle, size_t len)
{
	size_t i;
	size_t n;

	i = 0;
	n = 0;
	while (i < len && haystack[i+n] != '\0')
	{
		while (needle[n] == haystack[i+n] && needle[n] != '\0')
		{
			n++;
		}
		if (needle[n] == '\0')
		{
			return (((char *)&haystack[i]));
		}
		i++;
	}
	return (NULL);
}

int main (void)
{
    char *str = "hello, world, hello";
    char *word = "world";
    printf("%s\n",ft_strnstr(str, word, 13));
}
