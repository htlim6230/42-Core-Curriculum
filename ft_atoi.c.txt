int	ft_isspace(char *str)
{
	while(str[i] == ' ' || str[i] == '\n' || str[i] == '\r' || str[i] == '\t' || str[i] == '\v' || str[i] == '\f')
	{
		i++;
	}
	return (i);
}

int	ft_atoi(char *str)
{
	int i;
	int sign;
	int result;

	i = 0;
	sign = 1;
	result = 0;
	i = ft_isspace(str);

	while(str[i] == '+' || str[i] == '-')
	{
		if(str[i] == '-')
		{
			sign = -sign;
		}
		i++;
	}
	while(str[i] >= '0' && str[i] <= '9')
	{
		result = result*10 + (str[i] - '0');
		i++;
	}
	return (sign * result);
}
